====== ФАЙЛ: online.js ======
// Онлайн-индикатор (badge рядом с 🚚 в h2)
(function(){
 const g=id=>document.getElementById(id);
 const API=localStorage.getItem('ONLINE_API')||'https://eurobaza-online.example/api/online';
 let timer=null;

 function ensureBadge(){
  let b=g('onBadge');
  if(b)return b;
  const h=document.querySelector('h2 .badge');
  if(!h)return null;
  b=document.createElement('span');
  b.id='onBadge';
  b.className='online-badge';
  b.dataset.n='0';
  b.textContent='👥 0';
  h.after(b);
  return b;
 }

 async function ping(){
  const b=ensureBadge();
  if(!b)return;
  if(!navigator.onLine){b.dataset.n='0';b.textContent='👥 —';return}
  try{
   const r=await fetch(API+'?t='+Date.now(),{cache:'no-store',signal:AbortSignal.timeout(6000)});
   const j=await r.json();
   const n=(j&&typeof j.online==='number')?j.online:0;
   b.dataset.n=String(n);
   b.textContent='👥 '+n;
  }catch(e){
   b.dataset.n='0';
   b.textContent='👥 —';
  }
 }

 function start(){
  stop();
  ping();
  timer=setInterval(ping,30000);
 }
 function stop(){if(timer){clearInterval(timer);timer=null}}
 document.addEventListener('visibilitychange',()=>{document.hidden?stop():start()});
 window.addEventListener('online',start);
 window.addEventListener('offline',()=>{const b=ensureBadge();if(b){b.dataset.n='0';b.textContent='👥 —'}});
 start();
 window.onlineStart=start;
 window.onlineStop=stop;
})();


====== ФАЙЛ: chat.js ======
// Общий чат. Требует: <div id="chatMdl"> создаётся динамически,
// кнопка FAB .fab-c, поля #chatMsgList, #chatInput, #chatSend, #chatNameBtn
(function(){
 const g=id=>document.getElementById(id);
 const esc=t=>String(t).replace(/[&<>"]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]));
 const API=localStorage.getItem('CHAT_API')||'https://eurobaza-online.example/api/chat';
 let NAME=localStorage.getItem('CHAT_NAME')||'';
 let lastId=0, timer=null, open=false;

 function ensure(){
  let m=g('chatMdl');if(m)return m;
  m=document.createElement('div');
  m.id='chatMdl';m.className='mdl chat-mdl';
  m.innerHTML=''
   +'<div class="mdl-b chat-b">'
   +'<div class="mdl-h"><span>💬 <span id="chatTitle">Общий чат</span></span>'
   +'<button class="mdl-x" id="chatX">✕</button></div>'
   +'<div id="chatMsgList"></div>'
   +'<div class="chat-input-w">'
   +'<button id="chatNameBtn" title="Сменить имя">👤</button>'
   +'<input id="chatInput" placeholder="Сообщение...">'
   +'<button id="chatSend">➤</button>'
   +'</div></div>';
  document.body.appendChild(m);
  m.onclick=e=>{if(e.target===m)close()};
  g('chatX').onclick=()=>close();
  g('chatSend').onclick=send;
  g('chatInput').onkeydown=e=>{if(e.key==='Enter')send()};
  g('chatNameBtn').onclick=async()=>{
   const cur=NAME||'';
   const v=window.prompt?window.prompt('Ваше имя',cur):cur;
   if(v===null)return;
   const t=String(v).trim();
   if(!t){window.toast&&toast('Введите имя','err');return}
   NAME=t;
   localStorage.setItem('CHAT_NAME',NAME);
   window.toast&&toast('Имя сохранено');
  };
  return m;
 }

 async function load(initial){
  const list=g('chatMsgList');
  if(!list)return;
  if(!navigator.onLine){
   if(initial)list.innerHTML='<div class="bn-empty">Нет интернета</div>';
   return;
  }
  try{
   const url=API+'/list?after='+(initial?0:lastId)+'&t='+Date.now();
   const r=await fetch(url,{cache:'no-store',signal:AbortSignal.timeout(8000)});
   const j=await r.json();
   if(!Array.isArray(j.messages))throw 0;
   if(initial)list.innerHTML='';
   j.messages.forEach(m=>{
    if(typeof m.id==='number')lastId=Math.max(lastId,m.id);
    const mine=NAME&&m.author===NAME;
    const el=document.createElement('div');
    el.className='chat-msg'+(mine?' my-msg':'');
    el.innerHTML='<div class="chat-meta"><span class="chat-author">'+esc(m.author||'—')+'</span>'
     +'<span class="chat-time">'+esc(m.time||'')+'</span></div>'
     +'<div class="chat-text">'+esc(m.text||'')+'</div>';
    list.appendChild(el);
   });
   list.scrollTop=list.scrollHeight;
  }catch(e){
   if(initial){
    list.innerHTML='<div class="bn-empty">Не удалось загрузить чат</div>';
    window.toast&&toast('Не удалось загрузить чат','err');
   }
  }
 }

 async function send(){
  const inp=g('chatInput');
  if(!inp)return;
  const text=inp.value.trim();
  if(!text)return;
  if(!NAME){
   window.toast&&toast('Введите имя','err');
   g('chatNameBtn').click();
   return;
  }
  if(!navigator.onLine){window.toast&&toast('Нет интернета','err');return}
  inp.value='';
  try{
   const r=await fetch(API+'/send',{
    method:'POST',
    headers:{'Content-Type':'application/json'},
    body:JSON.stringify({author:NAME,text:text}),
    signal:AbortSignal.timeout(8000)
   });
   if(!r.ok)throw 0;
   load(false);
  }catch(e){
   window.toast&&toast('Ошибка: не отправлено','err');
  }
 }

 function openChat(){
  const m=ensure();
  m.classList.add('open');
  document.documentElement.classList.add('lock');
  document.body.classList.add('lock');
  open=true;
  try{history.pushState({chat:1},'')}catch(e){}
  load(true);
  timer=setInterval(()=>{if(!document.hidden)load(false)},5000);
 }
 function close(){
  const m=g('chatMdl');
  if(!m||!m.classList.contains('open'))return;
  m.classList.remove('open');
  document.documentElement.classList.remove('lock');
  document.body.classList.remove('lock');
  open=false;
  if(timer){clearInterval(timer);timer=null}
  try{if(history.state&&history.state.chat)setTimeout(()=>history.back(),0)}catch(e){}
 }
 window.addEventListener('popstate',()=>{if(open)close()});

 // FAB кнопка
 const fab=document.createElement('button');
 fab.className='fab fab-c';
 fab.id='chatFab';
 fab.textContent='💬';
 fab.onclick=openChat;
 document.body.appendChild(fab);

 window.openChat=openChat;
 window.closeChat=close;
})();


====== ФАЙЛ: sw.js ======
// Service Worker: кэш + обновление по version.json
const CACHE='eurobaza-v1';
const ASSETS=[
 './','./index.html','./style.css','./app.js','./yandex.js','./ui.js',
 './map.js','./bonus.js','./i18n.js','./online.js','./chat.js',
 './manifest.json','./icon-192.png'
];

self.addEventListener('install',e=>{
 self.skipWaiting();
 e.waitUntil(caches.open(CACHE).then(c=>c.addAll(ASSETS).catch(()=>{})));
});

self.addEventListener('activate',e=>{
 e.waitUntil(
  caches.keys().then(keys=>Promise.all(keys.filter(k=>k!==CACHE).map(k=>caches.delete(k))))
   .then(()=>self.clients.claim())
 );
});

self.addEventListener('message',e=>{
 if(e.data&&e.data.type==='SKIP_WAITING')self.skipWaiting();
});

self.addEventListener('fetch',e=>{
 const req=e.request;
 if(req.method!=='GET')return;
 const url=new URL(req.url);
 // version.json и API — только сеть
 if(url.pathname.endsWith('version.json')||url.pathname.startsWith('/api/')){
  e.respondWith(fetch(req).catch(()=>caches.match(req)));
  return;
 }
 e.respondWith(
  caches.match(req).then(hit=>{
   const net=fetch(req).then(res=>{
    if(res&&res.ok){
     const copy=res.clone();
     caches.open(CACHE).then(c=>c.put(req,copy)).catch(()=>{});
    }
    return res;
   }).catch(()=>hit);
   return hit||net;
  })
 );
});
